## Basic search

Search for a query and return the results:

<Tabs syncKey="integration">
<TabItem label="REST API">

```bash
curl -X POST \
  https://api.cloudflare.com/client/v4/accounts/{account_id}/websearch/search \
  -H "Authorization: Bearer {API_TOKEN}" \
  -H "Content-Type: application/json" \
  -d '{
    "query": "latest Cloudflare announcements",
    "limit": 5
  }'
```

</TabItem>
<TabItem label="Workers Binding">

<TypeScriptExample>

```ts
export default {
	async fetch(request, env): Promise<Response> {
		const { items } = await env.WEBSEARCH.search({
			query: "latest Cloudflare announcements",
			limit: 5,
		});

		return Response.json(
			items.map((r) => ({
				title: r.title,
				url: r.url,
				description: r.description,
			})),
		);
	},
} satisfies ExportedHandler<Env>;
```

</TypeScriptExample>

## Search and extract structured data

Search for URLs, then use [Browser Run /json](/browser-run/quick-actions/json-endpoint/) to extract structured data from the top results. Useful when you need specific fields from pages you do not know about upfront.

<Tabs syncKey="integration">
<TabItem label="REST API">

```bash
# 1. Search for relevant pages
RESULTS=$(curl -s -X POST \
  https://api.cloudflare.com/client/v4/accounts/{account_id}/websearch/search \
  -H "Authorization: Bearer {API_TOKEN}" \
  -H "Content-Type: application/json" \
  -d '{
    "query": "Cloudflare AI products",
    "limit": 5
  }')

# 2. Extract structured data from the first result using Browser Run /json
URL=$(echo $RESULTS | jq -r '.result.items[0].url')

curl -X POST \
  https://api.cloudflare.com/client/v4/accounts/{account_id}/browser-rendering/json \
  -H "Authorization: Bearer {API_TOKEN}" \
  -H "Content-Type: application/json" \
  -d '{
    "url": "'"$URL"'",
    "prompt": "Extract product information",
    "response_format": {
      "type": "json_schema",
      "schema": {
        "type": "object",
        "properties": {
          "product_name": { "type": "string" },
          "description": { "type": "string" },
          "use_case": { "type": "string" }
        }
      }
    }
  }'
```

</TabItem>
<TabItem label="Workers Binding">

<TypeScriptExample>

```ts
export default {
	async fetch(request, env): Promise<Response> {
		// 1. Search for relevant pages
		const { items } = await env.WEBSEARCH.search({
			query: "Cloudflare AI products",
			limit: 5,
		});

		// 2. Extract structured data from each result using Browser Run /json
		const extracted = await Promise.all(
			items.map(async (r) => {
				const response = await env.BROWSER.quickAction("json", {
					url: r.url,
					prompt: "Extract product information",
					response_format: {
						type: "json_schema",
						schema: {
							type: "object",
							properties: {
								product_name: { type: "string" },
								description: { type: "string" },
								use_case: { type: "string" },
							},
						},
					},
				});
				const data = await response.json();
				return { source: r.url, data };
			}),
		);

		return Response.json(extracted);
	},
} satisfies ExportedHandler<Env>;
```

</TypeScriptExample>

</TabItem>
</Tabs>

## Search and get markdown

Search for URLs, then fetch the full page content as markdown using [Browser Run /markdown](/browser-run/quick-actions/markdown-endpoint/). Good for feeding content into an LLM.

<Tabs syncKey="integration">
<TabItem label="REST API">

```bash
# 1. Search for relevant pages
RESULTS=$(curl -s -X POST \
  https://api.cloudflare.com/client/v4/accounts/{account_id}/websearch/search \
  -H "Authorization: Bearer {API_TOKEN}" \
  -H "Content-Type: application/json" \
  -d '{
    "query": "how to deploy Cloudflare Workers",
    "limit": 5
  }')

# 2. Get markdown for the first result using Browser Run /markdown
URL=$(echo $RESULTS | jq -r '.result.items[0].url')

curl -X POST \
  https://api.cloudflare.com/client/v4/accounts/{account_id}/browser-rendering/markdown \
  -H "Authorization: Bearer {API_TOKEN}" \
  -H "Content-Type: application/json" \
  -d '{
    "url": "'"$URL"'"
  }'
```

</TabItem>
<TabItem label="Workers Binding">

<TypeScriptExample>

```ts
export default {
	async fetch(request, env): Promise<Response> {
		const { items } = await env.WEBSEARCH.search({
			query: "how to deploy Cloudflare Workers",
			limit: 5,
		});

		const pages = await Promise.all(
			items.map(async (r) => {
				const response = await env.BROWSER.quickAction("markdown", {
					url: r.url,
				});
				const content = await response.text();
				return { title: r.title, url: r.url, content };
			}),
		);

		return Response.json(pages);
	},
} satisfies ExportedHandler<Env>;
```

</TypeScriptExample>

</TabItem>

    "query": "Cloudflare status page",
